============================================================ # MÀU GIAO DIỆN # ============================================================ BG = "#0b1020" PANEL = "#111a2e" CARD = "#17233d" CARD2 = "#1d2c4a" WHITE = "#f8fafc" MUTED = "#94a3b8" CYAN = "#38bdf8" GREEN = "#22c55e" RED = "#ef4444" YELLOW = "#facc15" # ============================================================ # XÓA GIAO DIỆN # ============================================================ def clear_root(): for widget in root.winfo_children(): widget.destroy() # ============================================================ # TẠO NÚT # ============================================================ def make_button( parent, text, command, bg=CARD2, fg=WHITE, width=20, height=2, font=15 ): return tk.Button( parent, text=text, command=command, font=("Segoe UI", font, "bold"), width=width, height=height, bg=bg, fg=fg, activebackground=bg, activeforeground=fg, relief="flat", bd=0, cursor="hand2" ) # ============================================================ # CHỌN MÀN # ============================================================ def choose_stage(stage): global selected_stage selected_stage = stage start_game() # ============================================================ # GIAO DIỆN GAME # ============================================================ def build_game_ui(): clear_root() root.configure( bg=BG ) # ---------------- HEADER ---------------- header = tk.Frame( root, bg="#101a31", height=82 ) header.pack( fill="x" ) header.pack_propagate(False) tk.Label( header, text="🧠 TOÁN 9", font=("Segoe UI", 24, "bold"), bg="#101a31", fg=WHITE ).pack( side="left", padx=28 ) stage_color = STAGES[selected_stage]["color"] tk.Label( header, text=( f"🎮 MÀN {selected_stage}: " f"{STAGES[selected_stage]['name']}" ), font=("Segoe UI", 13, "bold"), bg=stage_color, fg="#07111f", padx=15, pady=7 ).pack( side="left", padx=10 ) make_button( header, "☰ MENU", show_menu, bg="#273449", fg=WHITE, width=10, height=1, font=10 ).pack( side="right", padx=25 ) # ---------------- THÔNG SỐ ---------------- stats = tk.Frame( root, bg=BG ) stats.pack( fill="x", padx=45, pady=16 ) global score_label global question_label global timer_label global streak_label score_label = tk.Label( stats, text="⭐ 0 điểm", font=("Segoe UI", 14, "bold"), bg=BG, fg=YELLOW ) score_label.pack( side="left" ) question_label = tk.Label( stats, text="📚 Câu 1", font=("Segoe UI", 14, "bold"), bg=BG, fg=WHITE ) question_label.pack( side="left", padx=50 ) streak_label = tk.Label( stats, text="🔥 Chuỗi: 0", font=("Segoe UI", 14, "bold"), bg=BG, fg="#fb923c" ) streak_label.pack( side="right", padx=35 ) timer_label = tk.Label( stats, text="⏱️ 60s", font=("Segoe UI", 14, "bold"), bg=BG, fg=GREEN ) timer_label.pack( side="right" ) # ---------------- THANH THỜI GIAN ---------------- global progress_canvas progress_canvas = tk.Canvas( root, height=8, bg="#1e293b", highlightthickness=0 ) progress_canvas.pack( fill="x", padx=45 ) # ---------------- CARD ---------------- card = tk.Frame( root, bg=CARD ) card.pack( fill="both", expand=True, padx=45, pady=18 ) global question_text global message global answer_frame global answer_buttons global start_button question_text = tk.Label( card, text="", font=("Segoe UI", 25, "bold"), bg=CARD, fg=WHITE, wraplength=850, justify="center" ) question_text.pack( pady=(38, 20) ) # ---------------- ĐÁP ÁN ---------------- answer_frame = tk.Frame( card, bg=CARD ) answer_frame.pack() answer_buttons = [] for i in range(4): button = make_button( answer_frame, f"{chr(65 + i)}.", lambda i=i: answer( game_questions[ question_number - 1 ][1][i] ), bg=CARD2, fg=WHITE, width=29, height=2, font=14 ) button.grid( row=i // 2, column=i % 2, padx=10, pady=9 ) answer_buttons.append( button ) # ---------------- THÔNG BÁO ---------------- message = tk.Label( root, text="", font=("Segoe UI", 15, "bold"), bg=BG, fg=WHITE ) message.pack( pady=2 ) # ---------------- NÚT CHƠI LẠI ---------------- start_button = make_button( root, "🔄 CHƠI LẠI", start_game, bg=CYAN, fg="#07111f", width=16, height=1, font=12 ) start_button.pack( pady=(3, 14) ) # ============================================================ # CẬP NHẬT THÔNG SỐ # ============================================================ def update_stats(): score_label.config( text=f"⭐ {score} điểm" ) question_label.config( text=( f"📚 Câu {question_number}" f"/{len(game_questions)}" ) ) timer_label.config( text=f"⏱️ {time_left}s" ) streak_label.config( text=f"🔥 Chuỗi: {streak}" ) update_progress() # ============================================================ # THANH THỜI GIAN # ============================================================ def update_progress(): if not game_questions: return ratio = max( 0, min( 1, time_left / STAGES[selected_stage]["time"] ) ) progress_canvas.delete( "all" ) width = progress_canvas.winfo_width() if ratio > 0.5: color = GREEN elif ratio > 0.25: color = YELLOW else: color = RED progress_canvas.create_rectangle( 0, 0, width * ratio, 8, fill=color, outline="" ) # ============================================================ # CÂU HỎI MỚI # ============================================================ def new_question(): global current_answer if question_number > len(game_questions): end_game() return q = game_questions[ question_number - 1 ] current_answer = q[2] question_text.config( text=q[0] ) message.config( text="" ) for i, button in enumerate(answer_buttons): button.config( text=f"{chr(65+i)}. {q[1][i]}", state="normal", bg=CARD2, fg=WHITE ) # ============================================================ # HIỆU ỨNG ĐÚNG / SAI # ============================================================ def flash_effect(color): old_color = question_text.cget( "bg" ) question_text.config( bg=color ) root.after( 150, lambda: question_text.config( bg=old_color ) ) def animate_score(): old_font = score_label.cget( "font" ) score_label.config( font=("Segoe UI", 20, "bold") ) root.after( 220, lambda: score_label.config( font=old_font ) ) # ============================================================ # TRẢ LỜI # ============================================================ def answer(choice): global score global correct global wrong global streak if not running: return # Khóa nút để không trả lời 2 lần for button in answer_buttons: button.config( state="disabled" ) if choice == current_answer: # Tăng chuỗi trả lời đúng streak += 1 correct += 1 # Thưởng thêm điểm khi có chuỗi bonus = 0 if streak >= 3: bonus = (streak - 2) * 2 score += bonus score += STAGES[selected_stage]["points"] message.config( text=( "✨ CHÍNH XÁC! " f"+{STAGES[selected_stage]['points']} điểm " f"🔥 Chuỗi {streak}" + (f" (+{bonus} bonus)" if bonus else "") ), fg=GREEN ) flash_effect( "#173b35" ) animate_score() else: # Sai thì phá chuỗi streak = 0 wrong += 1 message.config( text=( "❌ CHƯA ĐÚNG!\n" f"Đáp án: {current_answer}" ), fg=RED ) flash_effect( "#451d29" ) update_stats() root.after( 700, next_question ) # ============================================================ # CÂU TIẾP THEO # ============================================================ def next_question(): global question_number if not running: return question_number += 1 if question_number > len(game_questions): end_game() else: new_question() update_stats() # ============================================================ # BẮT ĐẦU MÀN # ============================================================ def start_game(): global score global question_number global time_left global correct global wrong global streak global running global game_questions stage = STAGES[ selected_stage ] score = 0 correct = 0 wrong = 0 streak = 0 question_number = 1 time_left = stage["time"] running = True # Chọn câu ngẫu nhiên game_questions = random.sample( questions, min( stage["count"], len(questions) ) ) build_game_ui() new_question() update_stats() countdown() # ============================================================ # ĐẾM NGƯỢC # ============================================================ def countdown(): global time_left global running if not running: return if time_left <= 0: end_game() return time_left -= 1 update_stats() root.after( 1000, countdown ) # ============================================================ # KẾT THÚC MÀN # ============================================================ def end_game(): global running global unlocked_stage running = False for button in answer_buttons: button.config( state="disabled" ) # Tính tỷ lệ if game_questions: percent = ( correct / len(game_questions) ) * 100 else: percent = 0 # Mở màn mới nếu đạt 50% if percent >= 50: if selected_stage < len(STAGES): unlocked_stage = max( unlocked_stage, selected_stage + 1 ) save_progress() # Xếp hạng if percent >= 80: result = "🏆 XUẤT SẮC!" result_color = YELLOW elif percent >= 60: result = "👏 RẤT TỐT!" result_color = GREEN elif percent >= 50: result = "👍 ĐÃ VƯỢT QUA!" result_color = CYAN else: result = "💪 HÃY THỬ LẠI!" result_color = RED question_text.config( text=( f"{result}\n\n" f"🎮 Màn {selected_stage}: " f"{STAGES[selected_stage]['name']}\n\n" f"⭐ Điểm: {score}\n" f"✅ Đúng: {correct}/" f"{len(game_questions)}\n" f"❌ Sai: {wrong}\n" f"🔥 Chuỗi cuối: {streak}\n" f"📊 Tỷ lệ đúng: " f"{percent:.0f}%" ) ) message.config( text=( "🎉 Bạn đã hoàn thành màn!" if percent >= 50 else "😢 Chưa đạt 50%. Hãy thử lại!" ), fg=result_color ) # Nếu đã mở màn tiếp theo if ( selected_stage < len(STAGES) and unlocked_stage > selected_stage ): start_button.config( text=( f"➡️ MÀN " f"{selected_stage + 1}" ), command=lambda: choose_stage( selected_stage + 1 ) ) else: start_button.config( text="🔄 CHƠI LẠI", command=start_game ) # ============================================================ # CHẠY GAME # ============================================================ show_menu() root.mainloop()
